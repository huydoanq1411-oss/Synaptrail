31 Bài tập OOP & Collections

7 bài tập vận dụng class, interface, Collections và Stream API — mức độ thường gặp trong bài test tuyển dụng Junior Java Developer.

Bài 1: Class ChongDoi (Stack) tự cài đặt bằng ArrayList
class MyStack<T> {
    private List<T> duLieu = new ArrayList<>();

    public void push(T x) { duLieu.add(x); }
    public T pop() {
        if (duLieu.isEmpty()) throw new RuntimeException("Stack rỗng");
        return duLieu.remove(duLieu.size() - 1);
    }
    public T top() { return duLieu.get(duLieu.size() - 1); }
    public boolean isEmpty() { return duLieu.isEmpty(); }
}
Bài 2: Đếm tần suất từ bằng HashMap
static Map<String, Integer> demTu(String[] tuList) {
    Map<String, Integer> dem = new HashMap<>();
    for (String tu : tuList) {
        dem.put(tu, dem.getOrDefault(tu, 0) + 1);
    }
    return dem;
}
Bài 3: Interface HinhHoc với nhiều class triển khai
interface HinhHoc {
    double dienTich();
    double chuVi();
}

class HinhChuNhat implements HinhHoc {
    private double dai, rong;
    public HinhChuNhat(double d, double r) { dai = d; rong = r; }
    public double dienTich() { return dai * rong; }
    public double chuVi() { return 2 * (dai + rong); }
}
Bài 4: Two Sum bằng HashMap (O(n))
static int[] twoSum(int[] nums, int target) {
    Map<Integer, Integer> daThay = new HashMap<>();
    for (int i = 0; i < nums.length; i++) {
        int conLai = target - nums[i];
        if (daThay.containsKey(conLai)) {
            return new int[]{daThay.get(conLai), i};
        }
        daThay.put(nums[i], i);
    }
    return new int[]{};
}

Bài 6: Lớp Singleton (Design Pattern)
class KetNoiDatabase {
    private static KetNoiDatabase instance;
    private KetNoiDatabase() {}                 // constructor private

    public static KetNoiDatabase getInstance() {
        if (instance == null) {
            instance = new KetNoiDatabase();
        }
        return instance;
    }
}
