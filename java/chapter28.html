QuanLySinhVien.java 
import java.util.*;
import java.util.stream.Collectors;

class SinhVien {
    private String maSV, ten;
    private double diem;

    public SinhVien(String maSV, String ten, double diem) {
        this.maSV = maSV; this.ten = ten; this.diem = diem;
    }
    public String getTen() { return ten; }
    public double getDiem() { return diem; }
    @Override
    public String toString() {
        return maSV + " - " + ten + " - " + diem;
    }
}

public class QuanLySinhVien {
    private List<SinhVien> danhSach = new ArrayList<>();

    public void themSinhVien(SinhVien sv) {
        danhSach.add(sv);
    }

    public List<SinhVien> laySinhVienGioi() {              // dùng Stream API (chương 19)
        return danhSach.stream()
            .filter(sv -> sv.getDiem() >= 8.0)
            .sorted((a, b) -> Double.compare(b.getDiem(), a.getDiem()))
            .collect(Collectors.toList());
    }

    public double tinhDiemTrungBinh() {
        return danhSach.stream()
            .mapToDouble(SinhVien::getDiem)
            .average()
            .orElse(0);
    }

    public static void main(String[] args) {
        QuanLySinhVien ql = new QuanLySinhVien();
        ql.themSinhVien(new SinhVien("SV001", "Huy", 8.5));
        ql.themSinhVien(new SinhVien("SV002", "Lan", 9.2));
        ql.themSinhVien(new SinhVien("SV003", "Nam", 6.5));

        ql.laySinhVienGioi().forEach(System.out::println);
        System.out.printf("Điểm TB: %.2f%n", ql.tinhDiemTrungBinh());
    }
}
