30 Bài tập Cơ bản (Dễ)

8 bài tập củng cố cú pháp nền tảng, vòng lặp, method và mảng.

Bài 1: Kiểm tra số nguyên tố
static boolean laSoNguyenTo(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
Bài 2: Đảo ngược một số nguyên
static int daoNguoc(int n) {
    int ketQua = 0;
    while (n != 0) {
        ketQua = ketQua * 10 + n % 10;
        n /= 10;
    }
    return ketQua;
}
Bài 3: Kiểm tra chuỗi Palindrome
static boolean laPalindrome(String s) {
    String daoNguoc = new StringBuilder(s).reverse().toString();
    return s.equals(daoNguoc);
}
Bài 4: Tìm phần tử lớn nhất trong mảng
static int timMax(int[] mang) {
    int lonNhat = mang[0];
    for (int x : mang) {
        if (x > lonNhat) lonNhat = x;
    }
    return lonNhat;
}
Bài 5: Tính giai thừa bằng đệ quy
static long giaiThua(int n) {
    if (n <= 1) return 1;
    return n * giaiThua(n - 1);
}

Bài 7: Kiểm tra năm nhuận
static boolean laNamNhuan(int nam) {

Bài 8: Sắp xếp mảng bằng Bubble Sort
static void bubbleSort(int[] mang) {
    for (int i = 0; i < mang.length - 1; i++) {
        for (int j = 0; j < mang.length - i - 1; j++) {
            if (mang[j] > mang[j + 1]) {
                int tam = mang[j];
                mang[j] = mang[j + 1];
                mang[j + 1] = tam;
            }
        }
    }
}
