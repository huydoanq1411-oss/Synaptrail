static boolean laSoNguyenTo(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
static int daoNguoc(int n) {
    int ketQua = 0;
    while (n != 0) {
        ketQua = ketQua * 10 + n % 10;
        n /= 10;
    }
    return ketQua;
}
static boolean laPalindrome(String s) {
    String daoNguoc = new StringBuilder(s).reverse().toString();
    return s.equals(daoNguoc);
}
static int timMax(int[] mang) {
    int lonNhat = mang[0];
    for (int x : mang) {
        if (x > lonNhat) lonNhat = x;
    }
    return lonNhat;
}
static long giaiThua(int n) {
    if (n <= 1) return 1;
    return n * giaiThua(n - 1);
}
static int demNguyenAm(String s) {
    int dem = 0;
    String nguyenAm = "aeiouAEIOU";
    for (char c : s.toCharArray()) {
        if (nguyenAm.indexOf(c) != -1) dem++;
    }
    return dem;
}
static boolean laNamNhuan(int nam) {
    return (nam % 4 == 0 && nam % 100 != 0) || (nam % 400 == 0);
}
static void bubbleSort(int[] mang) {
    for (int i = 0; i < mang.length - 1; i++) {
        for (int j = 0; j < mang.length - i - 1; j++) {
            if (mang[j] > mang[j + 1]) {
                int tam = mang[j];
                mang[j] = mang[j + 1];
                mang[j + 1] = tam;
            }
        }
    }
}
