Kết hợp dữ liệu từ nhiều bảng có quan hệ với nhau — kỹ năng SQL quan trọng nhất sau CRUD cơ bản.

join.sql 
CREATE TABLE don_hang (
    id INT AUTO_INCREMENT PRIMARY KEY,
    san_pham_id INT,
    so_luong INT,
    FOREIGN KEY (san_pham_id) REFERENCES san_pham(id)   -- khóa ngoại, liên kết 2 bảng
);

-- INNER JOIN: chỉ lấy dòng CÓ khớp ở cả 2 bảng
SELECT don_hang.id, san_pham.ten, don_hang.so_luong
FROM don_hang
INNER JOIN san_pham ON don_hang.san_pham_id = san_pham.id;

-- LEFT JOIN: lấy TẤT CẢ dòng bảng trái, kể cả không khớp bên phải (NULL nếu không có)
SELECT san_pham.ten, don_hang.so_luong
FROM san_pham
LEFT JOIN don_hang ON san_pham.id = don_hang.san_pham_id;

GROUP BY và hàm tổng hợp

group_by.sql 
-- Tổng số lượng đã bán của MỖI sản phẩm
SELECT san_pham.ten, SUM(don_hang.so_luong) AS tong_ban
FROM san_pham
JOIN don_hang ON san_pham.id = don_hang.san_pham_id
GROUP BY san_pham.id
HAVING SUM(don_hang.so_luong) > 10;    -- HAVING lọc SAU KHI đã group, khác WHERE (lọc TRƯỚC khi group)
Ghi chú
WHERE lọc dòng TRƯỚC khi nhóm (GROUP BY), HAVING lọc SAU khi đã tính toán tổng hợp — đây là lý do không thể dùng WHERE SUM(...)>10 mà phải dùng HAVING.

📌 Tổng kết chương 2

JOIN kết hợp dữ liệu nhiều bảng qua khóa ngoại. INNER JOIN chỉ lấy dòng khớp cả 2 bên, LEFT JOIN giữ toàn bộ bảng trái. GROUP BY + HAVING dùng để tính và lọc dữ liệu tổng hợp.
