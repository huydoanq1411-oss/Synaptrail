Định nghĩa "hình dạng" của object và viết code tổng quát hoạt động với nhiều kiểu dữ liệu.

Interface — định nghĩa cấu trúc object

interface.ts 
interface SinhVien {
    ten: string;
    tuoi: number;
    email?: string;              // dấu ? = thuộc tính TÙY CHỌN, không bắt buộc
    readonly maSV: string;         // readonly = không thể gán lại sau khi khởi tạo
}

function gioiThieu(sv: SinhVien): string {
    return `${sv.ten}, ${sv.tuoi} tuổi`;
}

const sv1: SinhVien = { ten: "Huy", tuoi: 20, maSV: "SV001" };   // email không bắt buộc
console.log(gioiThieu(sv1));

Generics — code tổng quát cho nhiều kiểu dữ liệu

generics.ts 
function phanTuDauTien<T>(ds: T[]): T {    // T là kiểu tổng quát, tự suy luận theo input
    return ds[0];
}

console.log(phanTuDauTien<number>([1, 2, 3]));        // 1 - T = number
console.log(phanTuDauTien<string>(["a", "b"]));          // "a" - T = string
console.log(phanTuDauTien([true, false]));                    // T tự suy luận = boolean, không cần chỉ định

interface HopChua<T> {          // interface cũng dùng được generics
    noiDung: T;
}
const hopSo: HopChua<number> = { noiDung: 42 };
