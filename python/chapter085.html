selenium_basic.py 
# pip install selenium
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys

driver = webdriver.Chrome()          # Selenium 4.6+ tự tải đúng driver phù hợp (Selenium Manager), không cần cài ChromeDriver thủ công
driver.get("https://example.com")

o_tim_kiem = driver.find_element(By.NAME, "q")   # tìm phần tử theo thuộc tính name
o_tim_kiem.send_keys("Python là gì")
o_tim_kiem.send_keys(Keys.RETURN)                    # nhấn Enter

nut_dang_nhap = driver.find_element(By.ID, "login-btn")
nut_dang_nhap.click()                                    # click chuột

print(driver.title)   # lấy tiêu đề trang hiện tại
driver.quit()            # luôn đóng trình duyệt khi xong việc
Ghi chú
Trước đây, dùng Selenium yêu cầu tự tải file ChromeDriver đúng phiên bản với trình duyệt Chrome rồi trỏ đường dẫn thủ công — đây là nguồn lỗi phổ biến nhất với người mới. Từ Selenium 4.6 trở đi, Selenium Manager tích hợp sẵn tự động phát hiện phiên bản trình duyệt và tải driver phù hợp, nên đoạn code trên chạy được ngay mà không cần bước cài driver riêng.
