Dự án 3: Quản lý thư viện (mượn/trả sách)

quan_ly_thu_vien.py 
class Sach:
    def __init__(self, ma_sach, ten, con_lai):
        self.ma_sach = ma_sach
        self.ten = ten
        self.con_lai = con_lai

class ThuVien:
    def __init__(self):
        self.sach = {}          # ma_sach -> Sach
        self.nguoi_muon = {}      # ma_sach -> list ten người mượn

    def them_sach(self, sach: Sach):
        self.sach[sach.ma_sach] = sach

    def muon_sach(self, ma_sach, ten_nguoi):
        s = self.sach.get(ma_sach)
        if not s or s.con_lai <= 0:
            print("Sách không còn để mượn!")
            return
        s.con_lai -= 1
        self.nguoi_muon.setdefault(ma_sach, []).append(ten_nguoi)
        print(f"{ten_nguoi} đã mượn '{s.ten}' thành công")

    def tra_sach(self, ma_sach, ten_nguoi):
        if ma_sach in self.nguoi_muon and ten_nguoi in self.nguoi_muon[ma_sach]:
            self.nguoi_muon[ma_sach].remove(ten_nguoi)
            self.sach[ma_sach].con_lai += 1
            print(f"{ten_nguoi} đã trả sách thành công")

tv = ThuVien()
tv.them_sach(Sach("S001", "Clean Code", 2))
tv.muon_sach("S001", "Huy")
tv.tra_sach("S001", "Huy")

Dự án 4: Quản lý bán hàng (giỏ hàng + hóa đơn)

quan_ly_ban_hang.py 
class SanPham:
    def __init__(self, ten, gia):
        self.ten = ten
        self.gia = gia

class GioHang:
    def __init__(self):
        self.mat_hang = []   # list các (SanPham, so_luong)

    def them(self, san_pham: SanPham, so_luong=1):
        self.mat_hang.append((san_pham, so_luong))

    def tong_tien(self):
        return sum(sp.gia * sl for sp, sl in self.mat_hang)

    def xuat_hoa_don(self):
        print("=" * 30)
        for sp, sl in self.mat_hang:
            print(f"{sp.ten} x{sl} = {sp.gia * sl:,}đ")
        print("=" * 30)
        print(f"TỔNG CỘNG: {self.tong_tien():,}đ")

gio = GioHang()
gio.them(SanPham("Bàn phím", 500000), 1)
gio.them(SanPham("Chuột", 200000), 2)
gio.xuat_hoa_don()
