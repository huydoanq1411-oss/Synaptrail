Đề: Cho một list từ, nhóm chúng theo độ dài vào dictionary.

from collections import defaultdict

def nhom_theo_do_dai(ds_tu):
    nhom = defaultdict(list)
    for tu in ds_tu:
        nhom[len(tu)].append(tu)
    return dict(nhom)

print(nhom_theo_do_dai(["Python", "Java", "C", "Go", "Rust"]))
# {6: ['Python'], 4: ['Java'], 1: ['C'], 2: ['Go'], 4: ['Rust']}

Đề: Xây dựng class PhanSo hỗ trợ cộng hai phân số và hiển thị dạng a/b.

from math import gcd

class PhanSo:
    def __init__(self, tu, mau):
        uoc_chung = gcd(tu, mau)
        self.tu = tu // uoc_chung
        self.mau = mau // uoc_chung

    def __add__(self, khac):
        tu_moi = self.tu * khac.mau + khac.tu * self.mau
        mau_moi = self.mau * khac.mau
        return PhanSo(tu_moi, mau_moi)

    def __str__(self):
        return f"{self.tu}/{self.mau}"

a = PhanSo(1, 2)
b = PhanSo(1, 3)
print(a + b)   # 5/6

from collections import Counter

def thong_ke_log(duong_dan_file):
    dem = Counter()
    with open(duong_dan_file, "r", encoding="utf-8") as f:
        for dong in f:
            loai = dong.split(":")[0].strip()
            dem[loai] += 1
    return dict(dem)

import re

def kiem_tra_email(email):
    mau = r"^[\w\.-]+@[\w\.-]+\.\w+$"
    return re.match(mau, email) is not None

from collections import deque

class HangDoi:
    def __init__(self):
        self._ds = deque()

    def them(self, gia_tri):
        self._ds.append(gia_tri)

    def lay_ra(self):
        if not self._ds:
            raise IndexError("Hàng đợi rỗng")
        return self._ds.popleft()

    def rong(self):
        return len(self._ds) == 0

hd = HangDoi()
hd.them(1); hd.them(2); hd.them(3)

def gioi_han_goi(so_lan_toi_da):
    def decorator(func):
        dem = {"so_lan": 0}
        def wrapper(*args, **kwargs):
            if dem["so_lan"] >= so_lan_toi_da:
                print("Đã đạt giới hạn số lần gọi!")
                return None
            dem["so_lan"] += 1
            return func(*args, **kwargs)
        return wrapper
    return decorator

@gioi_han_goi(2)
def chao():
    print("Xin chào!")

chao(); chao(); chao()   # chỉ in ra "Xin chào!" 2 lần
def tron_dict(d1, d2):
    ket_qua = d1.copy()
    for key, value in d2.items():
        ket_qua[key] = ket_qua.get(key, 0) + value
    return ket_qua

print(tron_dict({"a": 1, "b": 2}, {"b": 3, "c": 4}))
# {'a': 1, 'b': 5, 'c': 4}
import time
from contextlib import contextmanager

@contextmanager
def do_thoi_gian(ten_tac_vu):
    bat_dau = time.time()
    yield
    print(f"{ten_tac_vu} mất {time.time() - bat_dau:.4f}s")

with do_thoi_gian("Tính tổng"):
    tong = sum(range(1_000_000))
def phan_trang(danh_sach, trang, so_luong_moi_trang):
    bat_dau = (trang - 1) * so_luong_moi_trang
    ket_thuc = bat_dau + so_luong_moi_trang
    return {
        "du_lieu": danh_sach[bat_dau:ket_thuc],
        "trang_hien_tai": trang,
        "tong_so_trang": (len(danh_sach) + so_luong_moi_trang - 1) // so_luong_moi_trang
    }

ds = list(range(1, 26))
print(phan_trang(ds, trang=2, so_luong_moi_trang=10))
# {'du_lieu': [11..20], 'trang_hien_tai': 2, 'tong_so_trang': 3}
