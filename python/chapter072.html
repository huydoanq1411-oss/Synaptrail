ORM (Object-Relational Mapping) cho phép thao tác database bằng code Python thay vì viết SQL thuần.

sqlalchemy_basic.py 
# pip install sqlalchemy
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.orm import declarative_base, sessionmaker

Base = declarative_base()

class SinhVien(Base):            # class Python <-> bảng trong database
    __tablename__ = "sinh_vien"
    id = Column(Integer, primary_key=True)
    ten = Column(String)
    diem = Column(Float)

engine = create_engine("sqlite:///truong.db")
Base.metadata.create_all(engine)   # tạo bảng từ class định nghĩa ở trên

Session = sessionmaker(bind=engine)
session = Session()

# Thêm dữ liệu - dùng object Python, không cần viết SQL
sv_moi = SinhVien(ten="Huy", diem=8.5)
session.add(sv_moi)
session.commit()

# Truy vấn - viết bằng code Python
ket_qua = session.query(SinhVien).filter(SinhVien.diem > 8.0).all()
for sv in ket_qua:
    print(sv.ten, sv.diem)
Mẹo
ORM giúp code không phụ thuộc vào một loại database cụ thể — chỉ cần đổi create_engine() để chuyển từ SQLite sang PostgreSQL mà không cần sửa logic truy vấn.
Cú pháp SQLAlchemy 2.0 (hiện đại hơn)
Ví dụ trên vẫn chạy tốt (SQLAlchemy vẫn giữ tương thích ngược), nhưng từ bản 2.0 trở đi, cách viết được khuyến nghị dùng type hint rõ ràng qua DeclarativeBase, Mapped và mapped_column — giúp IDE gợi ý kiểu dữ liệu chính xác hơn: 
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column

class Base(DeclarativeBase):
    pass

class SinhVien(Base):
    __tablename__ = "sinh_vien"
    id: Mapped[int] = mapped_column(primary_key=True)
    ten: Mapped[str]
    diem: Mapped[float]

# Nên mở Session bằng context manager để tự động đóng kết nối
with Session(engine) as session:
    session.add(SinhVien(ten="Huy", diem=8.5))
    session.commit()
