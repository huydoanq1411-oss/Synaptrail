pandas_basic.py 
# pip install pandas
import pandas as pd

df = pd.DataFrame({
    "ten": ["Huy", "Lan", "Nam"],
    "tuoi": [20, 22, 19],
    "diem": [8.5, 9.2, 7.0]
})

print(df.head())               # xem 5 dòng đầu
print(df["diem"].mean())         # điểm trung bình
print(df[df["diem"] > 8])          # lọc sinh viên có điểm > 8
df["xep_loai"] = df["diem"].apply(lambda x: "Giỏi" if x >= 8 else "Khá")   # thêm cột mới

print(df.sort_values("diem", ascending=False))    # sắp xếp giảm dần theo điểm
print(df.groupby("xep_loai")["diem"].mean())          # nhóm và tính trung bình theo nhóm

df_tu_csv = pd.read_csv("sinh_vien.csv")   # đọc dữ liệu từ file CSV
df.to_csv("ket_qua.csv", index=False)         # ghi ra file CSV
Ghi chú
DataFrame là cấu trúc trung tâm của Pandas — về bản chất là một bảng 2 chiều với các cột có thể có kiểu dữ liệu khác nhau, tương tự bảng tính Excel hoặc bảng SQL.
