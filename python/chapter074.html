Model — định nghĩa bảng database

models.py 
from django.db import models

class SanPham(models.Model):
    ten = models.CharField(max_length=100)
    gia = models.DecimalField(max_digits=10, decimal_places=2)
    ngay_tao = models.DateTimeField(auto_now_add=True)

View — xử lý request

views.py 
from django.http import JsonResponse
from .models import SanPham

def danh_sach_san_pham(request):
    san_pham = list(SanPham.objects.values())   # ORM tự sinh câu SQL
    return JsonResponse(san_pham, safe=False)
