datetime & pathlib

Xử lý ngày giờ và đường dẫn file theo phong cách hiện đại, hướng đối tượng.

datetime

datetime_module.py 
from datetime import datetime, timedelta

bay_gio = datetime.now()
print(bay_gio)                                # 2026-07-10 14:30:00.123456
print(bay_gio.strftime("%d/%m/%Y %H:%M"))       # định dạng lại: 10/07/2026 14:30

ngay_mai = bay_gio + timedelta(days=1)            # cộng thêm 1 ngày
print(ngay_mai)

ngay_sinh = datetime.strptime("15/08/2005", "%d/%m/%Y")  # chuyển chuỗi -> datetime
tuoi = (bay_gio - ngay_sinh).days // 365
print(f"Tuổi: {tuoi}")

pathlib — thao tác đường dẫn hiện đại

pathlib_module.py 
from pathlib import Path

p = Path("data") / "sinh_vien.csv"   # nối đường dẫn bằng toán tử /
print(p)                                # data/sinh_vien.csv
print(p.exists())                        # kiểm tra tồn tại
print(p.suffix)                           # .csv - phần mở rộng
print(p.stem)                              # sinh_vien - tên không có đuôi

for file in Path(".").glob("*.py"):        # liệt kê tất cả file .py trong thư mục
    print(file)
Mẹo
pathlib (hướng đối tượng, hiện đại) thường được khuyến nghị hơn os.path (hàm rời rạc, kiểu cũ) trong code Python mới.

📌 Tổng kết chương 50

datetime xử lý ngày giờ, định dạng và tính toán khoảng cách thời gian. pathlib thao tác đường dẫn file theo phong cách hướng đối tượng, dễ đọc hơn os.path.
