104 Bài tập Khó & Thuật toán

8 bài tập nâng cao về thuật toán và cấu trúc dữ liệu — mức độ phỏng vấn kỹ thuật tại các công ty công nghệ.

Bài 1: Longest Substring Without Repeating Characters

Đề: Tìm độ dài chuỗi con dài nhất không có ký tự lặp lại (kỹ thuật Sliding Window).

def do_dai_chuoi_con_khong_lap(s):
    da_thay = {}
    trai = 0
    max_do_dai = 0
    for phai, ky_tu in enumerate(s):
        if ky_tu in da_thay and da_thay[ky_tu] >= trai:
            trai = da_thay[ky_tu] + 1     # thu hẹp cửa sổ trái
        da_thay[ky_tu] = phai
        max_do_dai = max(max_do_dai, phai - trai + 1)
    return max_do_dai

print(do_dai_chuoi_con_khong_lap("abcabcbb"))   # 3 ("abc")
Bài 2: Kiểm tra chuỗi ngoặc hợp lệ (Valid Parentheses)
def ngoac_hop_le(s):
    stack = []
    cap = {")": "(", "]": "[", "}": "{"}
    for ky_tu in s:
        if ky_tu in "([{":
            stack.append(ky_tu)
        elif ky_tu in cap:
            if not stack or stack.pop() != cap[ky_tu]:
                return False
    return len(stack) == 0

print(ngoac_hop_le("()[]{}"))    # True
print(ngoac_hop_le("(]"))          # False

Bài 4: Longest Common Subsequence (Dynamic Programming)
def lcs(s1, s2):
    n, m = len(s1), len(s2)
    dp = [[0]*(m+1) for _ in range(n+1)]
    for i in range(1, n+1):
        for j in range(1, m+1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    return dp[n][m]

print(lcs("ABCBDAB", "BDCABA"))   # 4 ("BCBA")
Bài 5: Kiểm tra Binary Tree có cân bằng không
def chieu_cao(node):
    if node is None:
        return 0
    return 1 + max(chieu_cao(node.trai), chieu_cao(node.phai))

def can_bang(node):
    if node is None:
        return True
    lech = abs(chieu_cao(node.trai) - chieu_cao(node.phai))
    return lech <= 1 and can_bang(node.trai) and can_bang(node.phai)
Bài 6: Số đảo ngược nhiều nhất (Merge Sort đếm inversion)

Đề: Đếm số cặp (i, j) với i < j nhưng ds[i] > ds[j] — ứng dụng thực tế: đo "độ hỗn loạn" của một danh sách.

def dem_inversion(ds):
    if len(ds) <= 1:
        return ds, 0
    giua = len(ds) // 2
    trai, dem_trai = dem_inversion(ds[:giua])
    phai, dem_phai = dem_inversion(ds[giua:])
    tron, dem_tron = tron_va_dem(trai, phai)
    return tron, dem_trai + dem_phai + dem_tron

def tron_va_dem(trai, phai):
    ket_qua, i, j, dem = [], 0, 0, 0
    while i < len(trai) and j < len(phai):
        if trai[i] <= phai[j]:
            ket_qua.append(trai[i]); i += 1
        else:
            ket_qua.append(phai[j]); j += 1
            dem += len(trai) - i     # mọi phần tử còn lại của trái đều > phai[j]
    return ket_qua + trai[i:] + phai[j:], dem

Bài 7: LRU Cache (Least Recently Used)

Đề: Cài đặt cache có giới hạn kích thước, tự động loại bỏ mục ít dùng gần đây nhất khi đầy.

from collections import OrderedDict

class LRUCache:
    def __init__(self, suc_chua):
        self.suc_chua = suc_chua
        self.cache = OrderedDict()

    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)   # đánh dấu vừa được dùng
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.suc_chua:
            self.cache.popitem(last=False)   # xóa mục cũ nhất (ít dùng nhất)

lru = LRUCache(2)
lru.put(1, "A"); lru.put(2, "B")
lru.get(1)          # dùng key 1 -> 1 trở thành "mới nhất"
lru.put(3, "C")      # cache đầy -> xóa key 2 (ít dùng nhất)
print(lru.get(2))     # -1 (đã bị xóa)
Bài 8: Sắp xếp topo (Topological Sort) cho lịch học môn

Đề: Cho danh sách môn học và các môn tiên quyết, sắp xếp thứ tự học hợp lệ.

def sap_xep_topo(so_mon, tien_quyet):
    do_thi = {i: [] for i in range(so_mon)}
    bac_vao = [0] * so_mon
    for mon, can_hoc_truoc in tien_quyet:
        do_thi[can_hoc_truoc].append(mon)
        bac_vao[mon] += 1

    hang_doi = [i for i in range(so_mon) if bac_vao[i] == 0]
    thu_tu = []
    while hang_doi:
        mon = hang_doi.pop(0)
        thu_tu.append(mon)
        for mon_ke in do_thi[mon]:
            bac_vao[mon_ke] -= 1
            if bac_vao[mon_ke] == 0:
                hang_doi.append(mon_ke)

    return thu_tu if len(thu_tu) == so_mon else []   # rỗng nếu có chu trình (không thể học được)

print(sap_xep_topo(4, [[1,0],[2,0],[3,1],[3,2]]))   # [0, 1, 2, 3]
