Kết nối MySQL

mysql_connect.py 
# Cài đặt trước: pip install mysql-connector-python
import mysql.connector

conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="mat_khau",
    database="cua_hang"
)
cur = conn.cursor()
cur.execute("SELECT * FROM san_pham WHERE gia < %s", (500000,))
for row in cur.fetchall():
    print(row)
conn.close()

Kết nối PostgreSQL

postgresql_connect.py 
# Cài đặt trước: pip install psycopg2-binary
import psycopg2

conn = psycopg2.connect(
    host="localhost",
    dbname="cua_hang",
    user="postgres",
    password="mat_khau"
)
cur = conn.cursor()
cur.execute("SELECT * FROM san_pham WHERE gia < %s", (500000,))
for row in cur.fetchall():
    print(row)
conn.close()
