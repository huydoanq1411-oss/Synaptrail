Ví dụ: Open/Closed Principle

open_closed.py 
# VI PHẠM - mỗi khi thêm hình mới phải SỬA hàm này
def tinh_dien_tich(hinh):
    if hinh.loai == "vuong":
        return hinh.canh ** 2
    elif hinh.loai == "tron":
        return 3.14 * hinh.ban_kinh ** 2
    # thêm hình mới -> phải sửa lại hàm này -> vi phạm "Closed for modification"

# TUÂN THỦ - mở rộng bằng cách thêm class mới, KHÔNG sửa code cũ
class Hinh(ABC):
    @abstractmethod
    def dien_tich(self): pass

class HinhTamGiac(Hinh):        # thêm hình mới chỉ cần thêm class, không sửa gì khác
    def __init__(self, day, cao):
        self.day, self.cao = day, cao
    def dien_tich(self):
        return 0.5 * self.day * self.cao
