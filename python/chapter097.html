Dự án 1: Máy tính đơn giản (Console Calculator)

may_tinh.py 
def cong(a, b): return a + b
def tru(a, b): return a - b
def nhan(a, b): return a * b
def chia(a, b):
    if b == 0:
        raise ValueError("Không thể chia cho 0")
    return a / b

PHEP_TOAN = {"+": cong, "-": tru, "*": nhan, "/": chia}

def chay_may_tinh():
    print("=== MÁY TÍNH ĐƠN GIẢN === (gõ 'q' để thoát)")
    while True:
        bieu_thuc = input("Nhập phép tính (vd: 5 + 3): ")
        if bieu_thuc.lower() == "q":
            break
        try:
            a, toan_tu, b = bieu_thuc.split()
            ket_qua = PHEP_TOAN[toan_tu](float(a), float(b))
            print(f"Kết quả: {ket_qua}")
        except (ValueError, KeyError) as e:
            print(f"Lỗi: nhập sai định dạng hoặc phép toán không hỗ trợ ({e})")

if __name__ == "__main__":
    chay_may_tinh()

Dự án 2: Quản lý sinh viên (OOP + File JSON)

quan_ly_sinh_vien.py 
import json
from dataclasses import dataclass, asdict

@dataclass
class SinhVien:
    ma_sv: str
    ten: str
    diem: float

class QuanLySinhVien:
    def __init__(self, file_du_lieu="sinh_vien.json"):
        self.file_du_lieu = file_du_lieu
        self.danh_sach = self._doc_du_lieu()

    def _doc_du_lieu(self):
        try:
            with open(self.file_du_lieu, "r", encoding="utf-8") as f:
                return [SinhVien(**sv) for sv in json.load(f)]
        except FileNotFoundError:
            return []

    def luu_du_lieu(self):
        with open(self.file_du_lieu, "w", encoding="utf-8") as f:
            json.dump([asdict(sv) for sv in self.danh_sach], f, ensure_ascii=False, indent=2)

    def them_sinh_vien(self, sv: SinhVien):
        self.danh_sach.append(sv)
        self.luu_du_lieu()

    def xep_loai(self):
        return sorted(self.danh_sach, key=lambda sv: sv.diem, reverse=True)

    def diem_trung_binh(self):
        if not self.danh_sach:
            return 0
        return sum(sv.diem for sv in self.danh_sach) / len(self.danh_sach)

ql = QuanLySinhVien()
ql.them_sinh_vien(SinhVien("SV001", "Huy", 8.5))
ql.them_sinh_vien(SinhVien("SV002", "Lan", 9.2))
for sv in ql.xep_loai():
    print(f"{sv.ten}: {sv.diem}")
print(f"Điểm trung bình lớp: {ql.diem_trung_binh():.2f}")
