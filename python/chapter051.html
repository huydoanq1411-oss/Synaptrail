shutil

shutil_module.py 
import shutil

shutil.copy("a.txt", "backup/a.txt")     # sao chép file
shutil.copytree("du_an", "du_an_backup")   # sao chép toàn bộ thư mục
shutil.move("a.txt", "data/a.txt")          # di chuyển file
shutil.rmtree("thu_muc_cu")                   # xóa toàn bộ thư mục (kể cả không rỗng)

logging

logging_module.py 
import logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")

logging.debug("Chi tiết debug")     # không hiện vì level mặc định là INFO
logging.info("Chương trình bắt đầu chạy")
logging.warning("Sắp hết bộ nhớ")
logging.error("Không kết nối được database")
Quan trọng
Trong ứng dụng thực tế/chuyên nghiệp, nên dùng logging thay vì print() — logging có mức độ ưu tiên (level), ghi kèm thời gian, và có thể xuất ra file thay vì chỉ hiện trên console.

shutil xử lý sao chép/di chuyển/xóa file và thư mục cấp cao. logging ghi nhật ký chương trình chuyên nghiệp, có mức độ ưu tiên rõ ràng.
