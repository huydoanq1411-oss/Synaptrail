Danh sách liên kết — mỗi phần tử (node) chứa dữ liệu và con trỏ trỏ tới phần tử kế tiếp.

linked_list.py 
class Node:
    def __init__(self, du_lieu):
        self.du_lieu = du_lieu
        self.tiep_theo = None    # con trỏ tới node kế tiếp

class LinkedList:
    def __init__(self):
        self.dau = None            # node đầu tiên (head)

    def them_cuoi(self, du_lieu):
        node_moi = Node(du_lieu)
        if not self.dau:
            self.dau = node_moi
            return
        hien_tai = self.dau
        while hien_tai.tiep_theo:
            hien_tai = hien_tai.tiep_theo
        hien_tai.tiep_theo = node_moi

    def in_danh_sach(self):
        hien_tai = self.dau
        while hien_tai:
            print(hien_tai.du_lieu, end=" -> ")
            hien_tai = hien_tai.tiep_theo
        print("None")

ll = LinkedList()
ll.them_cuoi(1); ll.them_cuoi(2); ll.them_cuoi(3)
ll.in_danh_sach()   # 1 -> 2 -> 3 -> None
