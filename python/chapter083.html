Thư viện xử lý ảnh và thị giác máy tính (computer vision) phổ biến nhất.

opencv_basic.py 
# pip install opencv-python
import cv2

anh = cv2.imread("hinh.jpg")            # đọc ảnh
anh_xam = cv2.cvtColor(anh, cv2.COLOR_BGR2GRAY)   # chuyển sang ảnh xám
anh_resize = cv2.resize(anh, (300, 300))            # thay đổi kích thước

cv2.imwrite("hinh_xam.jpg", anh_xam)      # lưu ảnh đã xử lý

# Phát hiện khuôn mặt bằng mô hình có sẵn (Haar Cascade)
bo_phat_hien = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
khuon_mat = bo_phat_hien.detectMultiScale(anh_xam, scaleFactor=1.1, minNeighbors=5)
print(f"Tìm thấy {len(khuon_mat)} khuôn mặt")

for (x, y, w, h) in khuon_mat:
    cv2.rectangle(anh, (x, y), (x+w, y+h), (0, 255, 0), 2)   # vẽ khung quanh khuôn mặt
Mẹo
OpenCV lưu ảnh theo thứ tự kênh màu BGR (không phải RGB như hầu hết thư viện khác) — cần chuyển đổi bằng cv2.cvtColor(anh, cv2.COLOR_BGR2RGB) khi hiển thị bằng Matplotlib.
