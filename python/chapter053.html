Lập trình bất đồng bộ (asynchronous) — xử lý nhiều tác vụ I/O cùng lúc trên một luồng duy nhất.

asyncio_module.py 
import asyncio

async def tai_du_lieu(ten, giay):
    print(f"Bắt đầu tải {ten}")
    await asyncio.sleep(giay)   # giả lập chờ mạng - KHÔNG chặn các task khác
    print(f"Hoàn thành tải {ten}")
    return f"Dữ liệu từ {ten}"

async def main():
    # chạy đồng thời (concurrently), không phải tuần tự
    ket_qua = await asyncio.gather(
        tai_du_lieu("API A", 2),
        tai_du_lieu("API B", 1),
        tai_du_lieu("API C", 3),
    )
    print(ket_qua)

asyncio.run(main())   # tổng thời gian ~3s thay vì 2+1+3=6s nếu chạy tuần tự
Khái niệmÝ nghĩa
async defKhai báo một coroutine (hàm bất đồng bộ)
awaitChờ một coroutine khác hoàn thành mà không chặn toàn bộ chương trình
asyncio.run()Chạy coroutine chính (entry point)
asyncio.gather()Chạy nhiều coroutine đồng thời và chờ tất cả hoàn thành
Mẹo
asyncio đặc biệt mạnh trong các ứng dụng web hiện đại (FastAPI) và khi gọi nhiều API cùng lúc — thay vì chờ tuần tự từng cái, tất cả được chờ song song, tiết kiệm thời gian đáng kể.
