Chương 99 — Dự án: Web API, Web Scraper & AI Chatbot

Dự án 5: Web API quản lý công việc (FastAPI)

todo_api.py 
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()
cong_viec_db = {}
dem_id = 0

class CongViec(BaseModel):
    tieu_de: str
    hoan_thanh: bool = False

@app.post("/cong-viec")
def tao_cong_viec(cv: CongViec):
    global dem_id
    dem_id += 1
    cong_viec_db[dem_id] = cv
    return {"id": dem_id, **cv.dict()}

@app.get("/cong-viec/{id}")
def lay_cong_viec(id: int):
    if id not in cong_viec_db:
        raise HTTPException(status_code=404, detail="Không tìm thấy công việc")
    return cong_viec_db[id]

Dự án 6: Web Scraper lấy giá sản phẩm

scraper_gia.py 
import requests
from bs4 import BeautifulSoup
import csv

def lay_gia_san_pham(danh_sach_url):
    ket_qua = []
    for url in danh_sach_url:
        res = requests.get(url)
        soup = BeautifulSoup(res.text, "html.parser")
        ten = soup.find("h1", class_="ten-san-pham").text.strip()
        gia = soup.find("span", class_="gia-ban").text.strip()
        ket_qua.append({"ten": ten, "gia": gia, "url": url})
    return ket_qua

du_lieu = lay_gia_san_pham(["https://example.com/sp1", "https://example.com/sp2"])
with open("gia_san_pham.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=["ten", "gia", "url"])
    writer.writeheader()
    writer.writerows(du_lieu)

Dự án 7: AI Chatbot đơn giản (rule-based + ML)

chatbot_don_gian.py 
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

cau_hoi_mau = ["giờ mở cửa là gì", "làm sao để đặt hàng", "chính sách đổi trả"]
cau_tra_loi = ["Chúng tôi mở cửa 8h-22h hàng ngày", "Bạn có thể đặt hàng qua website", "Đổi trả trong 7 ngày"]

vectorizer = TfidfVectorizer()
vector_cau_hoi = vectorizer.fit_transform(cau_hoi_mau)

def tra_loi(cau_hoi_nguoi_dung):
    vector_nguoi_dung = vectorizer.transform([cau_hoi_nguoi_dung])
    diem_tuong_dong = cosine_similarity(vector_nguoi_dung, vector_cau_hoi)
    chi_so_gan_nhat = diem_tuong_dong.argmax()
    return cau_tra_loi[chi_so_gan_nhat]

print(tra_loi("mấy giờ các bạn mở cửa vậy"))   # Chúng tôi mở cửa 8h-22h hàng ngày
