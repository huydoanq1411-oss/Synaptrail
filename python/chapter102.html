Đề: Viết hàm nhận vào một số nguyên, trả về "Chẵn" hoặc "Lẻ".

def kiem_tra_chan_le(n):
    return "Chẵn" if n % 2 == 0 else "Lẻ"

print(kiem_tra_chan_le(7))   # Lẻ

Đề: Kiểm tra một số có phải số nguyên tố hay không.

def la_so_nguyen_to(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

print(la_so_nguyen_to(17))   # True
print(la_so_nguyen_to(15))   # False

Đề: Đảo ngược các chữ số của một số nguyên (VD: 123 -> 321).

def dao_nguoc_so(n):
    return int(str(n)[::-1])

print(dao_nguoc_so(123))   # 321

Đề: Kiểm tra một chuỗi có đọc xuôi/ngược giống nhau không (VD: "level").

def la_palindrome(chuoi):
    chuoi = chuoi.lower().replace(" ", "")
    return chuoi == chuoi[::-1]

print(la_palindrome("level"))     # True
print(la_palindrome("Python"))    # False

Đề: Tính n! không dùng đệ quy.

def giai_thua(n):
    ket_qua = 1
    for i in range(2, n + 1):
        ket_qua *= i
    return ket_qua

print(giai_thua(5))   # 120
def dem_nguyen_am(chuoi):
    nguyen_am = "aeiouAEIOU"
    return sum(1 for ky_tu in chuoi if ky_tu in nguyen_am)

print(dem_nguyen_am("Xin chao Python"))   # 4
def tim_max(ds):
    lon_nhat = ds[0]
    for so in ds:
        if so > lon_nhat:
            lon_nhat = so
    return lon_nhat

print(tim_max([3, 7, 2, 9, 4]))   # 9
def loai_trung_lap(ds):
    da_thay = set()
    ket_qua = []
    for x in ds:
        if x not in da_thay:
            da_thay.add(x)
            ket_qua.append(x)
    return ket_qua

print(loai_trung_lap([1, 2, 2, 3, 1, 4]))   # [1, 2, 3, 4]
def tam_giac_sao(n):
    for i in range(1, n + 1):
        print("*" * i)

def la_nam_nhuan(nam):
    return (nam % 4 == 0 and nam % 100 != 0) or (nam % 400 == 0)

print(la_nam_nhuan(2024))   # True
print(la_nam_nhuan(2100))   # False
def tong_chu_so(n):
    return sum(int(cs) for cs in str(n))

print(tong_chu_so(12345))   # 15
from collections import Counter
def dem_tan_suat(ds):
    return dict(Counter(ds))

print(dem_tan_suat([1,2,2,3,3,3]))   # {1: 1, 2: 2, 3: 3}

Đề: Hai chuỗi là anagram nếu chứa cùng các ký tự với cùng tần suất (VD: "listen" và "silent").

def la_anagram(s1, s2):
    return sorted(s1.replace(" ","").lower()) == sorted(s2.replace(" ","").lower())

print(la_anagram("listen", "silent"))   # True
def fibonacci_n(n):
    a, b = 0, 1
    ket_qua = []
    for _ in range(n):
        ket_qua.append(a)
        a, b = b, a + b
    return ket_qua

print(fibonacci_n(8))   # [0, 1, 1, 2, 3, 5, 8, 13]
