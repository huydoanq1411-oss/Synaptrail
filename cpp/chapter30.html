30 Dự án: Cấu trúc dữ liệu Ma trận

Dự án tổng hợp OOP + operator overloading — xây dựng class Ma trận hỗ trợ các phép toán cơ bản.

ma_tran.cpp 
class MaTran {
private:
    vector<vector<int>> duLieu;
    int soHang, soCot;
public:
    MaTran(int h, int c) : soHang(h), soCot(c) {
        duLieu = vector<vector<int>>(h, vector<int>(c, 0));
    }

    void dat(int hang, int cot, int giaTri) {
        duLieu[hang][cot] = giaTri;
    }

    int lay(int hang, int cot) const {
        return duLieu[hang][cot];
    }

    MaTran operator+(const MaTran& khac) const {        // overload phép cộng ma trận
        MaTran ketQua(soHang, soCot);
        for (int i = 0; i < soHang; i++)
            for (int j = 0; j < soCot; j++)
                ketQua.dat(i, j, duLieu[i][j] + khac.lay(i, j));
        return ketQua;
    }

    void in() const {
        for (int i = 0; i < soHang; i++) {
            for (int j = 0; j < soCot; j++) cout << duLieu[i][j] << " ";
            cout << endl;
        }
    }
};

int main() {
    MaTran a(2, 2), b(2, 2);
    a.dat(0,0,1); a.dat(0,1,2); a.dat(1,0,3); a.dat(1,1,4);
    b.dat(0,0,5); b.dat(0,1,6); b.dat(1,0,7); b.dat(1,1,8);

    MaTran c = a + b;    // dùng operator+ đã overload
    c.in();                 // 6 8 / 10 12
}
