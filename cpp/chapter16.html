Chương 16 — vector & string (STL)

STL (Standard Template Library) cung cấp các cấu trúc dữ liệu mạnh mẽ, an toàn hơn mảng C-style.

vector_basic.cpp 
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    vector<int> ds = {5, 2, 8, 1, 9};    // mảng ĐỘNG - tự thay đổi kích thước

    ds.push_back(10);          // thêm vào cuối - O(1) amortized
    ds.pop_back();                // xóa phần tử cuối
    cout << ds.size() << endl;      // kích thước hiện tại

    sort(ds.begin(), ds.end());   // sắp xếp - cần #include <algorithm>

    for (int x : ds) {              // duyệt bằng range-based for
        cout << x << " ";
    }

    // vector 2 chiều
    vector<vector<int>> maTran(3, vector<int>(3, 0));   // ma trận 3x3 toàn số 0
}
string_stl.cpp 
#include <string>

string s = "Xin chao";
cout << s.length() << endl;             // độ dài chuỗi
cout << s.substr(0, 3) << endl;           // "Xin" - cắt chuỗi từ vị trí 0, độ dài 3
cout << (s.find("chao") != string::npos) << endl;   // kiểm tra chứa chuỗi con
s += " Python";                             // nối chuỗi bằng +=
cout << s << endl;                           // "Xin chao Python"
Quan trọng
vector nên là lựa chọn mặc định thay cho mảng C-style trong hầu hết trường hợp — tự động quản lý bộ nhớ, tự thay đổi kích thước, và có .size()/.at() giúp code an toàn hơn nhiều.

📌 Tổng kết chương 16

vector là mảng động an toàn, nên dùng thay mảng C-style trong code hiện đại. string của STL cung cấp đầy đủ phương thức xử lý chuỗi tiện lợi.
