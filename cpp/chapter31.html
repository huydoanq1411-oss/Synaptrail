Chương 31 — Bài tập Cơ bản (Dễ)

8 bài tập củng cố cú pháp nền tảng, vòng lặp, hàm và mảng.

Bài 1: Kiểm tra số nguyên tố
bool laSoNguyenTo(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
Bài 2: Tìm phần tử lớn thứ 2 trong mảng
int timLonThu2(vector<int>& ds) {
    int max1 = INT_MIN, max2 = INT_MIN;
    for (int x : ds) {
        if (x > max1) { max2 = max1; max1 = x; }
        else if (x > max2 && x != max1) max2 = x;
    }
    return max2;
}
Bài 3: Đảo ngược chuỗi tại chỗ
void daoNguoc(string& s) {
    int trai = 0, phai = s.length() - 1;
    while (trai < phai) {
        swap(s[trai], s[phai]);
        trai++; phai--;
    }
}
Bài 4: Tính dãy Fibonacci bằng vòng lặp
vector<long long> fibonacci(int n) {
    vector<long long> ds = {0, 1};
    for (int i = 2; i < n; i++) {
        ds.push_back(ds[i-1] + ds[i-2]);
    }
    return ds;
}
Bài 5: Kiểm tra chuỗi Palindrome
bool laPalindrome(string s) {
    int trai = 0, phai = s.length() - 1;
    while (trai < phai) {
        if (s[trai] != s[phai]) return false;
        trai++; phai--;
    }
    return true;
}
Bài 6: Đếm tần suất ký tự trong chuỗi
map<char, int> demKyTu(string s) {
    map<char, int> dem;
    for (char c : s) dem[c]++;
    return dem;
}
Bài 7: Tìm ước chung lớn nhất (GCD)
int ucln(int a, int b) {
    while (b != 0) {
        int tam = b;
        b = a % b;
        a = tam;
    }
    return a;
}
Bài 8: Loại bỏ phần tử trùng lặp khỏi vector
vector<int> loaiTrung(vector<int> ds) {
    sort(ds.begin(), ds.end());
    ds.erase(unique(ds.begin(), ds.end()), ds.end());   // unique() gom trùng lặp liền kề về cuối
    return ds;
}
