26 Linked List tự cài đặt

Xây dựng danh sách liên kết từ đầu bằng con trỏ — bài tập kinh điển để hiểu sâu quản lý bộ nhớ.

linked_list.cpp 
struct Node {
    int giaTri;
    Node* tiepTheo;
    Node(int gt) : giaTri(gt), tiepTheo(nullptr) {}
};

class LinkedList {
private:
    Node* dau = nullptr;
public:
    void themCuoi(int giaTri) {
        Node* nodeMoi = new Node(giaTri);
        if (!dau) { dau = nodeMoi; return; }
        Node* hienTai = dau;
        while (hienTai->tiepTheo) hienTai = hienTai->tiepTheo;
        hienTai->tiepTheo = nodeMoi;
    }

    void in() {
        Node* hienTai = dau;
        while (hienTai) {
            cout << hienTai->giaTri << " -> ";
            hienTai = hienTai->tiepTheo;
        }
        cout << "NULL" << endl;
    }

    ~LinkedList() {                       // destructor - PHẢI tự giải phóng từng node
        Node* hienTai = dau;
        while (hienTai) {
            Node* tam = hienTai;
            hienTai = hienTai->tiepTheo;
            delete tam;
        }
    }
};

int main() {
    LinkedList ll;
    ll.themCuoi(1); ll.themCuoi(2); ll.themCuoi(3);
    ll.in();   // 1 -> 2 -> 3 -> NULL
}   // destructor tự động chạy khi ll ra khỏi phạm vi, giải phóng toàn bộ node
Quan trọng
Vì mỗi node được cấp phát bằng new, class phải có destructor tự giải phóng toàn bộ node bằng delete — nếu không, mọi node sẽ bị rò rỉ bộ nhớ khi object LinkedList bị hủy.

📌 Tổng kết chương 26

Linked List tự cài đặt bằng struct Node và con trỏ, minh họa việc quản lý bộ nhớ động thủ công — destructor phải tự giải phóng từng node để tránh memory leak.
