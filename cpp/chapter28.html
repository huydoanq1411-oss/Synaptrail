28 Đệ quy & Chia để trị

Kỹ thuật giải bài toán bằng cách chia nhỏ thành bài toán con giống hệt cấu trúc — nền tảng của nhiều thuật toán hiệu quả.

recursion.cpp 
long long giaiThua(int n) {
    if (n <= 1) return 1;              // điều kiện dừng - BẮT BUỘC
    return n * giaiThua(n - 1);
}

// Merge Sort - ví dụ kinh điển của chia để trị (divide and conquer)
void tron(vector<int>& ds, int trai, int giua, int phai) {
    vector<int> tam;
    int i = trai, j = giua + 1;
    while (i <= giua && j <= phai) {
        if (ds[i] <= ds[j]) tam.push_back(ds[i++]);
        else tam.push_back(ds[j++]);
    }
    while (i <= giua) tam.push_back(ds[i++]);
    while (j <= phai) tam.push_back(ds[j++]);
    for (int k = 0; k < tam.size(); k++) ds[trai + k] = tam[k];
}

void mergeSort(vector<int>& ds, int trai, int phai) {
    if (trai >= phai) return;                    // điều kiện dừng
    int giua = trai + (phai - trai) / 2;
    mergeSort(ds, trai, giua);                       // chia nửa trái
    mergeSort(ds, giua + 1, phai);                      // chia nửa phải
    tron(ds, trai, giua, phai);                            // trộn lại (conquer)
}
Cảnh báo — Stack Overflow
Mỗi lần gọi đệ quy tốn thêm một "khung" trên call stack. Đệ quy quá sâu (VD: đệ quy không có điều kiện dừng đúng, hoặc n quá lớn) sẽ gây lỗi Segmentation fault do tràn stack — khác với Python báo lỗi rõ ràng RecursionError.

Đệ quy cần điều kiện dừng rõ ràng để tránh stack overflow. Chia để trị (divide and conquer) như Merge Sort chia bài toán lớn thành các phần nhỏ hơn, giải rồi kết hợp lại.
