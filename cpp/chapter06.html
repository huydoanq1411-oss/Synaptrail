Nhận dữ liệu từ người dùng và xuất dữ liệu ra màn hình bằng luồng (stream).

io.cpp 
#include <iostream>
using namespace std;

int main() {
    string ten;
    int tuoi;

    cout << "Nhập tên: ";
    cin >> ten;                 // đọc MỘT TỪ (dừng ở khoảng trắng)

    cout << "Nhập tuổi: ";
    cin >> tuoi;                  // cin tự động chuyển đổi sang đúng kiểu int

    cout << "Xin chào " << ten << ", " << tuoi << " tuổi" << endl;
    return 0;
}

Đọc cả dòng có khoảng trắng với getline

getline.cpp 
string ho_ten;
cout << "Nhập họ tên đầy đủ: ";
cin.ignore();                 // bỏ qua ký tự xuống dòng còn sót lại từ cin >> trước đó
getline(cin, ho_ten);           // đọc TOÀN BỘ dòng, kể cả khoảng trắng
cout << "Xin chào " << ho_ten << endl;
Quan trọng
cin >> chỉ đọc đến khoảng trắng đầu tiên — muốn đọc cả câu (VD: "Nguyễn Văn A") phải dùng getline(), và thường cần cin.ignore() trước đó để "dọn" ký tự xuống dòng còn sót lại.
