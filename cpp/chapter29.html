quan_ly_sinh_vien.cpp 
#include <iostream>
#include <vector>
#include <fstream>
#include <sstream>
using namespace std;

class SinhVien {
public:
    string maSV, ten;
    double diem;
    SinhVien(string ma, string t, double d) : maSV(ma), ten(t), diem(d) {}

    string toDong() const {                          // chuyển thành 1 dòng để ghi file CSV
        return maSV + "," + ten + "," + to_string(diem);
    }
};

class QuanLySinhVien {
private:
    vector<SinhVien> danhSach;
    string tenFile = "sinhvien.csv";
public:
    void themSinhVien(const SinhVien& sv) {
        danhSach.push_back(sv);
    }

    void luuFile() {
        ofstream file(tenFile);
        for (const auto& sv : danhSach) {
            file << sv.toDong() << "\n";
        }
        file.close();
    }

    void docFile() {
        ifstream file(tenFile);
        string dong;
        while (getline(file, dong)) {
            stringstream ss(dong);
            string ma, ten, diemStr;
            getline(ss, ma, ',');
            getline(ss, ten, ',');
            getline(ss, diemStr, ',');
            danhSach.push_back(SinhVien(ma, ten, stod(diemStr)));
        }
    }

    void inDanhSach() {
        for (const auto& sv : danhSach) {
            cout << sv.maSV << " - " << sv.ten << " - " << sv.diem << endl;
        }
    }
};

int main() {
    QuanLySinhVien ql;
    ql.themSinhVien(SinhVien("SV001", "Huy", 8.5));
    ql.themSinhVien(SinhVien("SV002", "Lan", 9.2));
    ql.luuFile();
    ql.inDanhSach();
}
