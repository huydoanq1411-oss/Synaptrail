bool laAnagram(string s1, string s2) {
    if (s1.length() != s2.length()) return false;
    unordered_map<char, int> dem;
    for (char c : s1) dem[c]++;
    for (char c : s2) dem[c]--;
    for (auto& cap : dem) if (cap.second != 0) return false;
    return true;
}
vector<int> twoSum(vector<int>& ds, int target) {
    unordered_map<int, int> daThay;    // giá trị -> chỉ số, tra cứu O(1)
    for (int i = 0; i < ds.size(); i++) {
        int conLai = target - ds[i];
        if (daThay.count(conLai)) return {daThay[conLai], i};
        daThay[ds[i]] = i;
    }
    return {};
}
