Chương 14 — Đa hình & Hàm ảo (Virtual)

Cho phép gọi đúng phương thức của class con thông qua con trỏ/tham chiếu class cha — nền tảng đa hình trong C++.

virtual.cpp 
class Hinh {
public:
    virtual double dienTich() {         // "virtual" - BẮT BUỘC để đa hình hoạt động đúng
        return 0;
    }
    virtual ~Hinh() {}                     // destructor ảo - QUAN TRỌNG khi có kế thừa + con trỏ
};

class HinhVuong : public Hinh {
private:
    double canh;
public:
    HinhVuong(double c) : canh(c) {}
    double dienTich() override {         // "override" giúp trình biên dịch kiểm tra đúng chữ ký
        return canh * canh;
    }
};

class HinhTron : public Hinh {
private:
    double banKinh;
public:
    HinhTron(double r) : banKinh(r) {}
    double dienTich() override {
        return 3.14 * banKinh * banKinh;
    }
};

int main() {
    Hinh* danhSach[2];
    danhSach[0] = new HinhVuong(4);
    danhSach[1] = new HinhTron(3);

    for (int i = 0; i < 2; i++) {
        cout << danhSach[i]->dienTich() << endl;   // gọi ĐÚNG phiên bản của từng class con
    }
    for (int i = 0; i < 2; i++) delete danhSach[i];    // giải phóng bộ nhớ đã cấp phát
}
Quan trọng
Nếu KHÔNG khai báo virtual, gọi phương thức thông qua con trỏ class cha sẽ luôn chạy phiên bản của class cha (gọi là "static binding") thay vì phiên bản đã override ở class con — đây là lỗi khái niệm rất phổ biến khi mới học đa hình trong C++.
Cảnh báo
Luôn khai báo destructor ảo (virtual ~Hinh()) cho class cha nếu có kế thừa — nếu không, delete qua con trỏ class cha sẽ KHÔNG gọi destructor của class con, gây rò rỉ bộ nhớ (memory leak).
