Tham số URL và query string

ChiTiet.jsx 
import { useParams, useSearchParams } from 'react-router-dom';

function ChiTiet() {
  const { id } = useParams();                  // /san-pham/42  ->  id = "42"
  const [qs, setQs] = useSearchParams();       // ?sort=gia
  const sort = qs.get('sort') ?? 'moi';
  return <p>Sản phẩm {id}, sắp xếp: {sort}</p>;
}

Bảo vệ route cần đăng nhập

RequireAuth.jsx 
import { Navigate, Outlet } from 'react-router-dom';

function RequireAuth({ daDangNhap }) {
  return daDangNhap ? <Outlet /> : <Navigate to="/dang-nhap" replace />;
}
📝 Ghi chú
Giá trị lấy từ useParams luôn là chuỗi. Hãy ép kiểu bằng Number(id) khi cần so sánh với số.

📌 Tổng kết chương 14

Dùng Outlet cho layout lồng nhau, useParams/useSearchParams đọc URL, và Navigate để chuyển hướng khi chưa đăng nhập.
