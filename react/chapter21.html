createSlice

demSlice.js 
import { createSlice } from '@reduxjs/toolkit';

const demSlice = createSlice({
  name: 'dem',
  initialState: { gia_tri: 0 },
  reducers: {
    tang: state => { state.gia_tri += 1; },      // RTK dùng Immer: được "viết như sửa trực tiếp"
    giam: state => { state.gia_tri -= 1; },
    dat:  (state, action) => { state.gia_tri = action.payload; },
  },
});

export const { tang, giam, dat } = demSlice.actions;
export default demSlice.reducer;

Dùng trong component

Dem.jsx 
import { useSelector, useDispatch } from 'react-redux';
import { tang, giam } from './demSlice';

function Dem() {
  const gt = useSelector(s => s.dem.gia_tri);
  const dispatch = useDispatch();
  return (
    <>
      <button onClick={() => dispatch(giam())}>-</button>
      <span>{gt}</span>
      <button onClick={() => dispatch(tang())}>+</button>
    </>
  );
}

Gọi API với createAsyncThunk

userSlice.js 
export const taiUsers = createAsyncThunk('users/tai', async () => {
  const res = await fetch('/api/users');
  if (!res.ok) throw new Error('Không tải được');
  return res.json();
});

const userSlice = createSlice({
  name: 'users',
  initialState: { ds: [], trangThai: 'rong', loi: null },
  reducers: {},
  extraReducers: b => b
    .addCase(taiUsers.pending,   s => { s.trangThai = 'dangTai'; })
    .addCase(taiUsers.fulfilled, (s, a) => { s.trangThai = 'xong'; s.ds = a.payload; })
    .addCase(taiUsers.rejected,  (s, a) => { s.trangThai = 'loi'; s.loi = a.error.message; }),
});
💡 Mẹo
Với dữ liệu từ server, RTK Query (đi kèm Redux Toolkit) lo sẵn cache và trạng thái tải, thường gọn hơn tự viết thunk.

📌 Tổng kết chương 21

createSlice gộp reducer và action, createAsyncThunk xử lý API với ba trạng thái pending/fulfilled/rejected, component dùng useSelector và useDispatch.
