Form đăng ký có validation

DangKy.jsx 
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';

const schema = z.object({
  email: z.string().email('Email không hợp lệ'),
  matKhau: z.string().min(8, 'Tối thiểu 8 ký tự'),
});

function DangKy() {
  const { register, handleSubmit, formState: { errors, isSubmitting } } =
    useForm({ resolver: zodResolver(schema) });

  const onSubmit = async data => { await guiLenServer(data); };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register('email')} placeholder="Email" />
      {errors.email && <p>{errors.email.message}</p>}
      <input type="password" {...register('matKhau')} placeholder="Mật khẩu" />
      {errors.matKhau && <p>{errors.matKhau.message}</p>}
      <button disabled={isSubmitting}>Đăng ký</button>
    </form>
  );
}
📌 Lưu ý
Validation phía client chỉ cải thiện trải nghiệm. Server vẫn phải kiểm tra lại dữ liệu vì client có thể bị qua mặt.

Cách này dùng input không kiểm soát (uncontrolled) nên gõ phím không làm cả form render lại, nhanh hơn nhiều so với tự quản lý từng ô bằng useState khi form lớn.
