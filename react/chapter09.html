Chương 9 — Custom Hooks — Tái sử dụng logic

Custom Hook là hàm bắt đầu bằng use, gom logic có state để dùng lại ở nhiều component.

Tạo hook useFetch

useFetch.js 
import { useState, useEffect } from 'react';

export function useFetch(url) {
  const [data, setData] = useState(null);
  const [dangTai, setDangTai] = useState(true);
  const [loi, setLoi] = useState(null);

  useEffect(() => {
    let huy = false;
    setDangTai(true);
    fetch(url)
      .then(r => r.json())
      .then(d => { if (!huy) setData(d); })
      .catch(e => { if (!huy) setLoi(e.message); })
      .finally(() => { if (!huy) setDangTai(false); });
    return () => { huy = true; };
  }, [url]);

  return { data, dangTai, loi };
}
Dung.jsx 
function Posts() {
  const { data, dangTai, loi } = useFetch('/api/posts');
  if (dangTai) return <p>Đang tải...</p>;
  if (loi) return <p>{loi}</p>;
  return data.map(p => <h3 key={p.id}>{p.title}</h3>);
}
📌 Quy tắc
Tên custom hook phải bắt đầu bằng use để React và linter nhận ra và kiểm tra quy tắc Hooks. Mỗi lần gọi hook có state riêng, không chia sẻ state giữa các component.

Ví dụ khác: useLocalStorage

useLocalStorage.js 
export function useLocalStorage(key, init) {
  const [v, setV] = useState(() => {
    const s = localStorage.getItem(key);
    return s ? JSON.parse(s) : init;
  });
  useEffect(() => localStorage.setItem(key, JSON.stringify(v)), [key, v]);
  return [v, setV];
}
