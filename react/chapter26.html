Chương 26 — Dự án cuối khóa: Ứng dụng React hoàn chỉnh

Hook trung tâm

useTodos.js 
import { useReducer, useEffect } from 'react';

function reducer(ds, a) {
  switch (a.type) {
    case 'them':  return [...ds, { id: Date.now(), ten: a.ten, xong: false }];
    case 'doi':   return ds.map(t => t.id === a.id ? { ...t, xong: !t.xong } : t);
    case 'xoa':   return ds.filter(t => t.id !== a.id);
    default: return ds;
  }
}

export function useTodos() {
  const [ds, dispatch] = useReducer(reducer, [], () =>
    JSON.parse(localStorage.getItem('todos') || '[]'));
  useEffect(() => localStorage.setItem('todos', JSON.stringify(ds)), [ds]);
  return { ds, dispatch };
}

Danh sách việc cần làm

Bước 1 — Dựng khung

Tạo dự án Vite + React, cài react-router-dom, chia thư mục như gợi ý, khai báo các route.

Bước 2 — Quản lý state

Viết useTodos, dùng Context để chia sẻ cho các trang.

Bước 3 — Form và validation

Dùng react-hook-form cho form thêm/sửa, báo lỗi khi tên rỗng.

Bước 4 — Hoàn thiện

Thêm Error Boundary, viết vài test với Testing Library, build và deploy lên Vercel hoặc Netlify.

💡 Mở rộng
Thay localStorage bằng API thật (Node.js + Express hoặc NestJS) và dùng TanStack Query để cache dữ liệu.
