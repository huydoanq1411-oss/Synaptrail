Hook trung tâm

useTodos.js 
import { useReducer, useEffect } from 'react';

function reducer(ds, a) {
  switch (a.type) {
    case 'them':  return [...ds, { id: Date.now(), ten: a.ten, xong: false }];
    case 'doi':   return ds.map(t => t.id === a.id ? { ...t, xong: !t.xong } : t);
    case 'xoa':   return ds.filter(t => t.id !== a.id);
    default: return ds;
  }
}

export function useTodos() {
  const [ds, dispatch] = useReducer(reducer, [], () =>
    JSON.parse(localStorage.getItem('todos') || '[]'));
  useEffect(() => localStorage.setItem('todos', JSON.stringify(ds)), [ds]);
  return { ds, dispatch };
}
