Chương 2 — ORM Prisma

ORM hiện đại, được ưa chuộng gần đây nhờ type-safety mạnh mẽ khi kết hợp với TypeScript.

terminal 
npm install prisma @prisma/client
npx prisma init                  # tạo file schema.prisma và .env
prisma/schema.prisma 
datasource db {
    provider = "mysql"
    url      = env("DATABASE_URL")
}

generator client {
    provider = "prisma-client-js"
}

model SanPham {                        // định nghĩa model ngay trong file schema riêng, KHÔNG lẫn vào code
    id       Int      @id @default(autoincrement())
    ten      String
    gia      Decimal
    soLuong  Int      @default(0)
    donHangs DonHang[]                    // quan hệ 1-nhiều với bảng DonHang
}
terminal 
npx prisma migrate dev --name init     # tạo bảng thật trong database dựa theo schema
npx prisma generate                       # sinh ra Prisma Client với TYPE ĐẦY ĐỦ cho TypeScript
su_dung.ts 
import { PrismaClient } from '@prisma/client';
const prisma = new PrismaClient();

const spMoi = await prisma.sanPham.create({ data: { ten: 'Áo thun', gia: 150000 } });
const tatCa = await prisma.sanPham.findMany();
const locGia = await prisma.sanPham.findMany({ where: { gia: { gt: 100000 } } });
await prisma.sanPham.update({ where: { id: 1 }, data: { gia: 180000 } });
await prisma.sanPham.delete({ where: { id: 1 } });
