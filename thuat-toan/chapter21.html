Binary Search Tree (BST)

Cây nhị phân với ràng buộc đặc biệt: trái luôn nhỏ hơn gốc, phải luôn lớn hơn gốc — cho phép tìm kiếm O(log n).

bst.py 
class BSTNode:
    def __init__(self, gia_tri):
        self.gia_tri = gia_tri
        self.trai = None
        self.phai = None

def them(node, gia_tri):
    if node is None:
        return BSTNode(gia_tri)
    if gia_tri < node.gia_tri:
        node.trai = them(node.trai, gia_tri)
    else:
        node.phai = them(node.phai, gia_tri)
    return node

def tim_kiem(node, gia_tri):
    if node is None or node.gia_tri == gia_tri:
        return node
    if gia_tri < node.gia_tri:
        return tim_kiem(node.trai, gia_tri)      # chỉ cần đi MỘT NHÁNH, không cần xét nhánh kia
    return tim_kiem(node.phai, gia_tri)

goc = None
for x in [5, 3, 8, 1, 4, 7, 9]:
    goc = them(goc, x)

print(tim_kiem(goc, 7) is not None)   # True

Xóa node trong BST — trường hợp phức tạp nhất

bst_xoa.py 
def tim_min(node):                   # tìm giá trị nhỏ nhất - luôn ở TẬN CÙNG BÊN TRÁI
    while node.trai:
        node = node.trai
    return node

def xoa(node, gia_tri):
    if node is None:
        return None
    if gia_tri < node.gia_tri:
        node.trai = xoa(node.trai, gia_tri)
    elif gia_tri > node.gia_tri:
        node.phai = xoa(node.phai, gia_tri)
    else:                                  # đã tìm thấy node cần xóa - 3 trường hợp:
        if node.trai is None:                # TH1: không có con trái -> thay bằng con phải
            return node.phai
        elif node.phai is None:                # TH2: không có con phải -> thay bằng con trái
            return node.trai
        else:                                    # TH3: có ĐỦ 2 con -> phức tạp nhất
            ke_thua = tim_min(node.phai)             # tìm giá trị nhỏ nhất bên nhánh phải
            node.gia_tri = ke_thua.gia_tri              # thay giá trị node bằng giá trị đó
            node.phai = xoa(node.phai, ke_thua.gia_tri)   # rồi xóa node kế thừa (giờ đã trùng)
    return node
Quan trọng
Độ phức tạp của BST (tìm kiếm/thêm/xóa) là O(log n) chỉ khi cây cân bằng. Nếu thêm dữ liệu đã sắp xếp sẵn vào BST thông thường, cây sẽ suy biến thành một danh sách liên kết — độ phức tạp tệ đi thành O(n). Đây là lý do cần AVL Tree (chương 22).

BST giữ ràng buộc trái-nhỏ/phải-lớn, cho phép tìm kiếm O(log n) khi cân bằng. Xóa node có đủ 2 con là trường hợp phức tạp nhất, cần tìm giá trị kế thừa (successor) để thay thế.
