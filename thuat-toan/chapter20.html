20 Binary Tree & Duyệt cây

Cấu trúc dữ liệu phân cấp — mỗi node có tối đa 2 con. Nền tảng cho BST, Heap, và nhiều cấu trúc nâng cao khác.

Cài đặt cơ bản

binary_tree.py 
class Node:
    def __init__(self, gia_tri):
        self.gia_tri = gia_tri
        self.trai = None
        self.phai = None

#         1
#       /   \\
#      2     3
#     / \\
#    4   5
goc = Node(1)
goc.trai = Node(2)
goc.phai = Node(3)
goc.trai.trai = Node(4)
goc.trai.phai = Node(5)

3 kiểu duyệt DFS (Depth-First)

duyet_cay.py 
def preorder(node, ket_qua=None):    # Gốc -> Trái -> Phải
    if ket_qua is None: ket_qua = []
    if node:
        ket_qua.append(node.gia_tri)
        preorder(node.trai, ket_qua)
        preorder(node.phai, ket_qua)
    return ket_qua

def inorder(node, ket_qua=None):     # Trái -> Gốc -> Phải
    if ket_qua is None: ket_qua = []
    if node:
        inorder(node.trai, ket_qua)
        ket_qua.append(node.gia_tri)
        inorder(node.phai, ket_qua)
    return ket_qua

def postorder(node, ket_qua=None):   # Trái -> Phải -> Gốc
    if ket_qua is None: ket_qua = []
    if node:
        postorder(node.trai, ket_qua)
        postorder(node.phai, ket_qua)
        ket_qua.append(node.gia_tri)
    return ket_qua

print(preorder(goc))    # [1, 2, 4, 5, 3]
print(inorder(goc))     # [4, 2, 5, 1, 3]
print(postorder(goc))   # [4, 5, 2, 3, 1]

Duyệt BFS (theo chiều rộng, từng tầng)

bfs_tree.py 
from collections import deque

def level_order(goc):          # duyệt theo TỪNG TẦNG, dùng Queue (chương 19)
    if not goc: return []
    ket_qua, hang_doi = [], deque([goc])
    while hang_doi:
        node = hang_doi.popleft()
        ket_qua.append(node.gia_tri)
        if node.trai: hang_doi.append(node.trai)
        if node.phai: hang_doi.append(node.phai)
    return ket_qua

print(level_order(goc))   # [1, 2, 3, 4, 5]
Mẹo ghi nhớ
Inorder có tính chất đặc biệt trên BST (chương 21): luôn cho kết quả đã sắp xếp. Postorder thường dùng khi cần xử lý node con trước khi xử lý node cha (VD: tính kích thước thư mục, giải phóng bộ nhớ cây).
