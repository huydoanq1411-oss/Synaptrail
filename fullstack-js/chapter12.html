12 ORM Sequelize

Object-Relational Mapping — thao tác database bằng object/class JavaScript thay vì viết SQL thuần tay.

terminal 
npm install sequelize mysql2
models/SanPham.js 
const { Sequelize, DataTypes } = require('sequelize');
const sequelize = new Sequelize('cua_hang', 'root', process.env.DB_PASSWORD, { dialect: 'mysql' });

const SanPham = sequelize.define('SanPham', {          // định nghĩa model TƯƠNG ỨNG với 1 bảng
    ten: { type: DataTypes.STRING, allowNull: false },
    gia: { type: DataTypes.DECIMAL(10, 2), allowNull: false },
    soLuong: { type: DataTypes.INTEGER, defaultValue: 0 },
});

module.exports = { sequelize, SanPham };
su_dung.js 
const { SanPham } = require('./models/SanPham');

// CREATE
const spMoi = await SanPham.create({ ten: 'Áo thun', gia: 150000 });

// READ
const tatCa = await SanPham.findAll();
const mot = await SanPham.findByPk(1);                             // tìm theo khóa chính
const locGia = await SanPham.findAll({ where: { gia: { [Op.gt]: 100000 } } });   // WHERE gia > 100000

// UPDATE
await SanPham.update({ gia: 180000 }, { where: { id: 1 } });

// DELETE
await SanPham.destroy({ where: { id: 1 } });
Mẹo
Sequelize tự động sinh câu SQL đúng chuẩn, ĐÃ chống SQL Injection sẵn — code trông giống thao tác object JavaScript thông thường, không cần tự viết chuỗi SQL nữa.
