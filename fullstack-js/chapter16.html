Chương 16 — Socket.IO — Giao tiếp Real-time

HTTP request-response truyền thống không phù hợp cho chat, thông báo tức thời — Socket.IO cho phép server "chủ động" gửi dữ liệu về client.

WebSocket là gì?

Khác với HTTP (client luôn phải hỏi trước, server mới trả lời), WebSocket duy trì một kết nối MỞ LIÊN TỤC giữa client-server, cho phép CẢ HAI BÊN chủ động gửi dữ liệu bất cứ lúc nào. Socket.IO là thư viện phổ biến nhất để dùng WebSocket dễ dàng trong Node.js.

terminal 
npm install socket.io          # phía server
npm install socket.io-client     # phía client (React/trình duyệt)
server.js 
const { Server } = require('socket.io');
const io = new Server(3001, { cors: { origin: '*' } });

io.on('connection', (socket) => {
    console.log('Người dùng đã kết nối:', socket.id);

    socket.on('gui_tin_nhan', (duLieu) => {             // lắng nghe sự kiện TỪ client
        console.log('Nhận tin nhắn:', duLieu);
        io.emit('tin_nhan_moi', duLieu);                    // gửi tới TẤT CẢ client đang kết nối
    });

    socket.on('disconnect', () => {
        console.log('Người dùng đã ngắt kết nối');
    });
});
ChatBox.jsx 
import { useEffect, useState } from 'react';
import { io } from 'socket.io-client';

const socket = io('http://localhost:3001');

function ChatBox() {
    const [tinNhans, setTinNhans] = useState([]);

    useEffect(() => {
        socket.on('tin_nhan_moi', (duLieu) => {           // lắng nghe tin nhắn MỚI từ server
            setTinNhans(truoc => [...truoc, duLieu]);
        });
        return () => socket.off('tin_nhan_moi');              // dọn dẹp listener khi component unmount
    }, []);

    const guiTinNhan = (noiDung) => {
        socket.emit('gui_tin_nhan', { noiDung, thoiGian: Date.now() });
    };

    return (/* giao diện hiển thị tinNhans và ô nhập liệu */ null);
}
Quan trọng
io.emit() gửi tới TẤT CẢ client. Để gửi RIÊNG cho một người (VD: thông báo cá nhân), dùng socket.emit() trên đúng socket đó, hoặc tổ chức theo "room" (socket.join()/io.to(room).emit()) để gửi cho một nhóm cụ thể.
