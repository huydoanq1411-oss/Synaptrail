Chương 11 — Kết nối Node.js với MySQL

Truy vấn database trực tiếp từ code Node.js/Express — bước đệm trước khi dùng ORM tiện lợi hơn ở chương sau.

terminal 
npm install mysql2
db.js 
const mysql = require('mysql2/promise');

const pool = mysql.createPool({            // connection pool - tái sử dụng kết nối, hiệu quả hơn tạo mới mỗi lần
    host: 'localhost',
    user: 'root',
    password: process.env.DB_PASSWORD,        // luôn đọc từ biến môi trường, không hardcode
    database: 'cua_hang',
});

module.exports = pool;
routes/sanPham.js 
const express = require('express');
const router = express.Router();
const pool = require('../db');

router.get('/san-pham', async (req, res) => {
    const [rows] = await pool.query('SELECT * FROM san_pham');
    res.json(rows);
});

router.post('/san-pham', async (req, res) => {
    const { ten, gia } = req.body;
    // DÙNG DẤU ? (prepared statement) - KHÔNG BAO GIỜ nối chuỗi trực tiếp
    const [ketQua] = await pool.query(
        'INSERT INTO san_pham (ten, gia) VALUES (?, ?)', [ten, gia]
    );
    res.status(201).json({ id: ketQua.insertId, ten, gia });
});

module.exports = router;
Quan trọng — SQL Injection
TUYỆT ĐỐI KHÔNG nối chuỗi trực tiếp để tạo câu SQL (VD: `SELECT * FROM users WHERE ten='${ten}'`) — kẻ tấn công có thể chèn SQL độc hại qua input. Luôn dùng dấu ? (prepared statement) như ví dụ trên để thư viện tự động escape dữ liệu an toàn.

📌 Tổng kết chương 15

mysql2 kết nối Node.js với MySQL qua connection pool. Luôn dùng prepared statement (dấu ?) thay vì nối chuỗi trực tiếp để tránh lỗ hổng SQL Injection nghiêm trọng.
