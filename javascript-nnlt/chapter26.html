26 Dự án: Todo App (React)

Xây dựng lại chính ứng dụng Todo App bằng React — so sánh trực tiếp để thấy sự khác biệt.

⚠️ Cần kiến thức React trước
Bài này giả định bạn đã biết JSX, Component, Props và State/useState. Lộ trình JavaScript (NNLT) không còn dạy React tại chỗ nữa — toàn bộ phần React (JSX, Props, State, Hooks, Router, Context...) nay nằm trong lộ trình riêng React.js Từ Zero Đến Hero. Nếu chưa học React, hãy hoàn thành lộ trình đó trước rồi quay lại bài này.
TodoApp.jsx 
import { useState } from 'react';

function TodoApp() {
    const [congViecs, setCongViecs] = useState([]);
    const [oNhap, setONhap] = useState('');

    function themCongViec() {
        if (oNhap.trim() === '') return;
        setCongViecs([...congViecs, { tieuDe: oNhap, hoanThanh: false }]);   // spread (chương 11)
        setONhap('');
    }

    function toggleHoanThanh(index) {
        const moi = [...congViecs];
        moi[index].hoanThanh = !moi[index].hoanThanh;
        setCongViecs(moi);
    }

    function xoaCongViec(index) {
        setCongViecs(congViecs.filter((_, i) => i !== index));    // filter (chương 10)
    }

    return (
        <div>
            <input value={oNhap} onChange={(e) => setONhap(e.target.value)} />
            <button onClick={themCongViec}>Thêm</button>
            <ul>
                {congViecs.map((cv, index) => (
                    <li key={index}
                        style={{ textDecoration: cv.hoanThanh ? 'line-through' : 'none' }}
                        onClick={() => toggleHoanThanh(index)}>
                        {cv.tieuDe}
                        <button onClick={(e) => { e.stopPropagation(); xoaCongViec(index); }}>
                            Xóa
                        </button>
                    </li>
                ))}
            </ul>
        </div>
    );
}

export default TodoApp;
Mẹo — Sự khác biệt cốt lõi
Với React, ta chỉ cần khai báo UI trông như thế nào dựa trên state (congViecs.map(...)) — không cần tự viết logic render() để cập nhật DOM thủ công như ở chương 28. React tự động tính toán và cập nhật đúng phần DOM cần thay đổi.
