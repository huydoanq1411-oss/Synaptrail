Chương 15 — async/await

Cú pháp hiện đại (ES2017) giúp code bất đồng bộ trông và đọc giống code đồng bộ thông thường.

async_await.js 
async function layThongTinSinhVien() {       // "async" đánh dấu hàm trả về Promise
    try {
        const sv = await taiSinhVien();          // "await" tạm dừng cho tới khi Promise resolve
        const diem = await taiDiem(sv);            // trông giống code đồng bộ, dễ đọc hơn .then() nhiều
        const xepLoai = await taiXepLoai(diem);
        console.log(xepLoai);
    } catch (loi) {                                    // try-catch xử lý lỗi, thay cho .catch()
        console.log("Lỗi:", loi);
    }
}

layThongTinSinhVien();
Quan trọng
await chỉ dùng được bên trong hàm khai báo async. Về bản chất, async/await chỉ là "lớp vỏ cú pháp" (syntactic sugar) cho Promise — bên dưới vẫn hoạt động dựa trên cơ chế Promise y hệt chương 14.

Chạy song song với async/await

parallel_await.js 
async function taiNhieuDuLieu() {
    // CHẬM - chờ tuần tự từng cái
    const a = await taiA();
    const b = await taiB();

    // NHANH HƠN - chạy song song bằng Promise.all
    const [x, y] = await Promise.all([taiA(), taiB()]);
}
