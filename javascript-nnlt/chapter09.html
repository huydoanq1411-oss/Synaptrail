09 Hàm (Function & Arrow Function)

JavaScript có nhiều cách khai báo hàm — Arrow Function (ES6+) là phong cách hiện đại phổ biến nhất.

function.js 
// Cách 1: Function declaration (kiểu truyền thống)
function cong(a, b) {
    return a + b;
}

// Cách 2: Function expression
const tru = function(a, b) {
    return a - b;
};

// Cách 3: Arrow function (ES6+) - NGẮN GỌN, phổ biến nhất hiện nay
const nhan = (a, b) => a * b;              // 1 dòng, tự động return
const chia = (a, b) => {                        // nhiều dòng, cần return tường minh
    if (b === 0) return null;
    return a / b;
};

console.log(cong(3, 5));   // 8
console.log(nhan(3, 5));     // 15

Tham số mặc định & Rest parameters

function_advanced.js 
function gioiThieu(ten, tuoi = 18) {   // giá trị mặc định
    console.log(`${ten}, ${tuoi} tuổi`);
}
gioiThieu("Lan");   // Lan, 18 tuổi

function tinhTong(...soList) {          // rest parameter - gom mọi tham số thành mảng
    return soList.reduce((tong, so) => tong + so, 0);
}
console.log(tinhTong(1, 2, 3, 4));   // 10
Quan trọng — Arrow Function và "this"
Arrow function không có this riêng — nó dùng this của phạm vi bao ngoài (lexical this). Đây là lý do arrow function không phù hợp làm method của object khi cần truy cập this của chính object đó.

📌 Tổng kết chương 9

Arrow function (=>) là cách viết hàm ngắn gọn, hiện đại nhất. Rest parameter (...) gom nhiều tham số thành mảng. Cẩn thận với cách arrow function xử lý this.
