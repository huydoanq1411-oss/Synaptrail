File System (fs module)

Đọc và ghi file trên server bằng module fs có sẵn của Node.js.

Đọc và ghi file trên server bằng module fs có sẵn của Node.js.

fs_basic.js 
const fs = require('fs');
const fsPromises = require('fs/promises');   // phiên bản dùng Promise, hiện đại hơn

// Cách CŨ - callback (dễ gây callback hell)
fs.readFile('data.txt', 'utf8', (err, data) => {
    if (err) { console.error(err); return; }
    console.log(data);
});

// Cách HIỆN ĐẠI - dùng async/await
async function docFile() {
    try {
        const noiDung = await fsPromises.readFile('data.txt', 'utf8');
        console.log(noiDung);

        await fsPromises.writeFile('output.txt', 'Nội dung mới');
    } catch (err) {
        console.error('Lỗi:', err.message);
    }
}
docFile();
Mẹo
Luôn ưu tiên phiên bản fs/promises kết hợp async/await thay vì callback API cũ — code sạch hơn nhiều và tránh được callback hell khi cần đọc/ghi nhiều file liên tiếp.

📌 Tổng kết chương 22
