Bài 1: Kiểm tra số nguyên tố
function laSoNguyenTo(n) {
    if (n < 2) return false;
    for (let i = 2; i * i <= n; i++) {
        if (n % i === 0) return false;
    }
    return true;
}

Bài 3: Kiểm tra Palindrome
function laPalindrome(s) {
    const sach = s.toLowerCase().replace(/[^a-z0-9]/g, '');
    return sach === sach.split('').reverse().join('');
}
Bài 4: Tìm phần tử xuất hiện nhiều nhất trong mảng
function phanTuNhieuNhat(ds) {
    const dem = {};
    let ketQua = ds[0], maxDem = 0;
    for (const x of ds) {
        dem[x] = (dem[x] || 0) + 1;
        if (dem[x] > maxDem) { maxDem = dem[x]; ketQua = x; }
    }
    return ketQua;
}
Bài 5: Debounce function (kỹ thuật quan trọng trong thực tế)

Đề: Viết hàm debounce - trì hoãn thực thi cho tới khi ngừng gọi trong một khoảng thời gian (hữu ích cho ô tìm kiếm, chống spam click).

function debounce(func, delay) {
    let timer;
    return function (...args) {
        clearTimeout(timer);
        timer = setTimeout(() => func(...args), delay);
    };
}

const timKiemDaDebounce = debounce((tuKhoa) => {
    console.log("Đang tìm kiếm:", tuKhoa);
}, 500);
Bài 6: Flatten mảng lồng nhau
function flatten(ds) {
    return ds.reduce((ket, x) =>
        Array.isArray(x) ? ket.concat(flatten(x)) : ket.concat(x), []);
}
console.log(flatten([1, [2, [3, 4], 5], 6]));   // [1, 2, 3, 4, 5, 6]
// Hoặc dùng có sẵn: ds.flat(Infinity)

function deepClone(obj) {
    return JSON.parse(JSON.stringify(obj));   // cách đơn giản (không xử lý được Date, function...)
}
