    if (n < 2) return false;

}

function laPalindrome(s) {
    const sach = s.toLowerCase().replace(/[^a-z0-9]/g, '');

function phanTuNhieuNhat(ds) {
    const dem = {};
    let ketQua = ds[0], maxDem = 0;
    for (const x of ds) {
        dem[x] = (dem[x] || 0) + 1;
        if (dem[x] > maxDem) { maxDem = dem[x]; ketQua = x; }
    }
    return ketQua;
}

function debounce(func, delay) {
    let timer;
    return function (...args) {
        clearTimeout(timer);
        timer = setTimeout(() => func(...args), delay);
    };
}

function flatten(ds) {
    return ds.reduce((ket, x) =>
        Array.isArray(x) ? ket.concat(flatten(x)) : ket.concat(x), []);
}
