todo_app.js 
// Giả sử HTML có: <input id="oNhap"> <button id="nutThem">Thêm</button> <ul id="danhSach"></ul>

let congViecs = [];

function render() {
    const ul = document.getElementById('danhSach');
    ul.innerHTML = '';                                  // xóa và vẽ lại toàn bộ danh sách
    congViecs.forEach((cv, index) => {
        const li = document.createElement('li');
        li.textContent = cv.tieuDe;
        li.style.textDecoration = cv.hoanThanh ? 'line-through' : 'none';

        li.addEventListener('click', () => {              // click để đánh dấu hoàn thành
            congViecs[index].hoanThanh = !congViecs[index].hoanThanh;
            render();
        });

        const nutXoa = document.createElement('button');
        nutXoa.textContent = 'Xóa';
        nutXoa.addEventListener('click', (e) => {
            e.stopPropagation();                              // ngăn sự kiện click lan lên li
            congViecs.splice(index, 1);
            render();
        });

        li.appendChild(nutXoa);
        ul.appendChild(li);
    });
}

document.getElementById('nutThem').addEventListener('click', () => {
    const oNhap = document.getElementById('oNhap');
    if (oNhap.value.trim() === '') return;
    congViecs.push({ tieuDe: oNhap.value, hoanThanh: false });
    oNhap.value = '';
    render();
});
Mẹo
Dự án này minh họa chính xác vấn đề mà React giải quyết: mỗi lần dữ liệu thay đổi, ta phải TỰ TAY cập nhật lại DOM (render()) — với ứng dụng phức tạp hơn, việc quản lý DOM thủ công này trở nên rất khó kiểm soát.
