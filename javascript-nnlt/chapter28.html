Bài 2: Two Sum bằng Map (O(n))
function twoSum(nums, target) {
    const daThay = new Map();
    for (let i = 0; i < nums.length; i++) {
        const conLai = target - nums[i];
        if (daThay.has(conLai)) return [daThay.get(conLai), i];
        daThay.set(nums[i], i);
    }
    return [];
}
Bài 3: Chuyển đổi Promise callback cũ sang async/await
function cho(ms) {                        // hàm tiện ích "chờ" - rất hay dùng
    return new Promise(resolve => setTimeout(resolve, ms));
}

async function inTuanTu() {
    for (let i = 1; i <= 3; i++) {
        await cho(1000);
        console.log(i);                       // in 1, 2, 3 mỗi số cách nhau 1 giây
    }
}

    const ketQua = await Promise.allSettled(urls.map(url => fetch(url)));

        r.status === 'fulfilled' ? r.value : { loi: r.reason }

Bài 5: Memoize function (cache kết quả hàm)
function memoize(func) {
    const cache = new Map();
    return function (...args) {
        const key = JSON.stringify(args);
        if (cache.has(key)) return cache.get(key);
        const ketQua = func(...args);
        cache.set(key, ketQua);
        return ketQua;
    };
}
Bài 6: Custom Hook đếm số (React)
function useCounter(giaTriBanDau = 0) {
    const [dem, setDem] = useState(giaTriBanDau);
    const tang = () => setDem(d => d + 1);
    const giam = () => setDem(d => d - 1);
    const resetVe = () => setDem(giaTriBanDau);
    return { dem, tang, giam, resetVe };
}
// Dùng: const { dem, tang, giam } = useCounter(0);
