Design Patterns thường gặp trong JS

Các khuôn mẫu giải pháp đã được kiểm chứng cho những vấn đề lập trình lặp đi lặp lại.

Module Pattern — đóng gói, tránh ô nhiễm biến toàn cục

module_pattern.js 
const BoDemModule = (function() {
    let dem = 0;                          // biến PRIVATE - không truy cập được từ bên ngoài

    return {                                 // chỉ những gì return ra mới là PUBLIC
        tang() { dem++; return dem; },
        layGiaTri() { return dem; },
    };
})();

BoDemModule.tang();
console.log(BoDemModule.layGiaTri());   // 1
// console.log(BoDemModule.dem);        // undefined - không truy cập trực tiếp được

Observer Pattern — nền tảng của Event System

observer_pattern.js 
class BoPhatSuKien {
    constructor() { this.nguoiNghe = {}; }

    dangKy(tenSuKien, ham) {                              // đăng ký "lắng nghe" một sự kiện
        if (!this.nguoiNghe[tenSuKien]) this.nguoiNghe[tenSuKien] = [];
        this.nguoiNghe[tenSuKien].push(ham);
    }

    phatSuKien(tenSuKien, duLieu) {                          // "phát" sự kiện, gọi TẤT CẢ hàm đã đăng ký
        (this.nguoiNghe[tenSuKien] || []).forEach(ham => ham(duLieu));
    }
}

const boPhat = new BoPhatSuKien();
boPhat.dangKy('donHangMoi', (duLieu) => console.log('Có đơn hàng mới:', duLieu));
boPhat.phatSuKien('donHangMoi', { id: 1, tongTien: 500000 });
Ghi chú
Observer Pattern chính là nguyên lý nền tảng của addEventListener trong DOM (đã học ở chương 12) và hệ thống sự kiện của Node.js (EventEmitter) — hiểu pattern này giúp hiểu sâu hơn cách các API quen thuộc hoạt động bên dưới.

Singleton Pattern — đảm bảo chỉ có 1 instance

singleton_pattern.js 
class KetNoiDatabase {
    static instance;

    constructor() {
        if (KetNoiDatabase.instance) {
            return KetNoiDatabase.instance;      // trả về instance CŨ nếu đã tồn tại
        }
        KetNoiDatabase.instance = this;
    }
}

const ket_noi1 = new KetNoiDatabase();
const ket_noi2 = new KetNoiDatabase();
console.log(ket_noi1 === ket_noi2);   // true - CÙNG một instance duy nhất
