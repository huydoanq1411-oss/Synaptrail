Xây dựng REST API hoàn chỉnh và hiểu khái niệm middleware — trái tim của Express.

rest_api.js 
const express = require('express');
const app = express();
app.use(express.json());

let sinhVienDB = [{id: 1, ten: "Huy"}];

app.get('/sinh-vien', (req, res) => {
    res.json(sinhVienDB);
});

app.post('/sinh-vien', (req, res) => {
    const svMoi = {id: sinhVienDB.length + 1, ten: req.body.ten};
    sinhVienDB.push(svMoi);
    res.status(201).json(svMoi);       // 201 = Created
});

app.delete('/sinh-vien/:id', (req, res) => {
    sinhVienDB = sinhVienDB.filter(sv => sv.id != req.params.id);
    res.status(204).send();               // 204 = No Content
});

Middleware — hàm chạy "ở giữa" request và response

middleware.js 
function ghiLog(req, res, next) {          // middleware TỰ VIẾT
    console.log(`${req.method} ${req.url} - ${new Date().toISOString()}`);
    next();                                        // BẮT BUỘC gọi next() để chuyển sang middleware/route kế tiếp
}

function kiemTraDangNhap(req, res, next) {
    const token = req.headers['authorization'];
    if (!token) {
        return res.status(401).json({loi: "Chưa đăng nhập"});   // KHÔNG gọi next() -> dừng lại tại đây
    }
    next();
}

app.use(ghiLog);                                    // áp dụng cho MỌI route
app.get('/admin', kiemTraDangNhap, (req, res) => {    // áp dụng cho MỘT route cụ thể
    res.send('Trang quản trị');
});
Quan trọng
Middleware là khái niệm cốt lõi của Express: mỗi request đi qua một "chuỗi" các hàm middleware trước khi tới route handler cuối cùng. Quên gọi next() sẽ khiến request bị "treo" mãi mãi, không bao giờ có phản hồi.

📌 Tổng kết chương 24

REST API trong Express định nghĩa qua các route GET/POST/PUT/DELETE. Middleware (qua app.use() hoặc tham số route) xử lý logic chung như logging, xác thực — luôn nhớ gọi next().
